const BASE = "/templates/datum";

export function assetSrc(key: string, width: 800 | 1600, format: "webp" | "avif" = "webp"): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

type PictureProps = {
  assetKey: string;
  alt: string;
  className?: string;
  sizes?: string;
  priority?: boolean;
};

/** Responsive <picture>, or a labelled optical placeholder when no raster asset is present. */
export function Picture({
  assetKey,
  alt,
  className,
  sizes = "(min-width: 900px) 44vw, 100vw",
  priority,
}: PictureProps) {
  const webp800 = assetSrc(assetKey, 800);
  if (!webp800) {
    return (
      <div className={`dt-placeholder ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>
        <span className="dt-mono">{assetKey}</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  const set = (format: "webp" | "avif") => `${assetSrc(assetKey, 800, format)} 800w, ${assetSrc(assetKey, 1600, format)} 1600w`;
  return (
    <picture className={className}>
      <source type="image/avif" srcSet={set("avif")} sizes={sizes} />
      <img
        src={webp800}
        srcSet={set("webp")}
        sizes={sizes}
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        fetchPriority={priority ? "high" : undefined}
        decoding="async"
        data-asset={assetKey}
      />
    </picture>
  );
}

<![CDATA[export function clamp(value: number, min: number, max: number): number {
  return Math.min(max, Math.max(min, value));
}

export function lerp(from: number, to: number, t: number): number {
  return from + (to - from) * t;
}

/** 0 at `start`, 1 at `end`, clamped; start === end acts as a step function. */
export function progress(value: number, start: number, end: number): number {
  if (end === start) return value >= end ? 1 : 0;
  return clamp((value - start) / (end - start), 0, 1);
}

/** Scroll progress of a tall section whose sticky viewport stage covers one screen. */
export function sectionProgress(
  scrollY: number,
  sectionTop: number,
  sectionHeight: number,
  viewportHeight: number,
): number {
  return progress(scrollY, sectionTop, sectionTop + Math.max(1, sectionHeight - viewportHeight));
}

/** Capped stagger delay in milliseconds so long lists never feel sluggish. */
export function staggerDelay(index: number, stepMs: number, capMs = 600): number {
  return Math.min(capMs, Math.max(0, index) * stepMs);

  as?: ElementType;
  className?: string;
  children: ReactNode;

/** Writes --rx, --ry, --mx, --my from rAF on fine pointers for perspective tilt and radial spotlight. */
export function TiltCard({ as: Tag = "div", className, children }: TiltCardProps) {
  const ref = useRef<HTMLElement>(null);
  const reduced = useReducedMotion();

  useEffect(() => {
    const node = ref.current;
    if (!node || reduced || !window.matchMedia("(hover: hover) and (pointer: fine)").matches) return;
    let frame = 0;
    let pending: PointerEvent | null = null;

    const apply = () => {
      frame = 0;
      if (!pending) return;
      const box = node.getBoundingClientRect();
      if (!box.width || !box.height) return;
      const px = (pending.clientX - box.left) / box.width;
      const py = (pending.clientY - box.top) / box.height;
      const { rotateX, rotateY } = tiltAngles(px, py, MAX_TILT_DEG);
      node.style.setProperty("--rx", `${rotateX.toFixed(2)}deg`);
      node.style.setProperty("--ry", `${rotateY.toFixed(2)}deg`);
      node.style.setProperty("--mx", `${(px * 100).toFixed(1)}%`);
      node.style.setProperty("--my", `${(py * 100).toFixed(1)}%`);
    };
    const onMove = (event: PointerEvent) => {
      pending = event;
      if (!frame) frame = requestAnimationFrame(apply);
    };
    const onLeave = () => {
      pending = null;
      node.style.setProperty("--rx", "0deg");
      node.style.setProperty("--ry", "0deg");
    };
    node.addEventListener("pointermove", onMove, { passive: true });
    node.addEventListener("pointerleave", onLeave, { passive: true });
    return () => {
      node.removeEventListener("pointermove", onMove);
      node.removeEventListener("pointerleave", onLeave);
      cancelAnimationFrame(frame);
    };
  }, [reduced]);

  return (
    <Tag ref={ref} className={className} data-tilt>
      {children}
    </Tag>
  );
}

import { useEffect, useRef, useState } from "react";

type Options = { once?: boolean; rootMargin?: string; threshold?: number };
type Entry = { callback: (inView: boolean) => void; once: boolean };

const observers = new Map<string, { observer: IntersectionObserver; entries: Map<Element, Entry> }>();

function sharedObserver(rootMargin: string, threshold: number) {
  const key = `${rootMargin}|${threshold}`;
  let record = observers.get(key);
  if (!record) {
    const entries = new Map<Element, Entry>();
    const observer = new IntersectionObserver(
      (batch) => {
        for (const item of batch) {
          const entry = entries.get(item.target);
          if (!entry) continue;
          entry.callback(item.isIntersecting);
          if (item.isIntersecting && entry.once) {
            entries.delete(item.target);
            observer.unobserve(item.target);
          }
        }
      },
      { rootMargin, threshold },
    );
    record = { observer, entries };
    observers.set(key, record);
  }
  return record;
}

/** Shares one IntersectionObserver instance per (rootMargin, threshold) pair across the template. */
export function useInView<T extends Element>({
  once = true,
  rootMargin = "0px 0px -10% 0px",
  threshold = 0,
}: Options = {}) {
  const ref = useRef<T>(null);
  const [inView, setInView] = useState(false);

  useEffect(() => {
    const node = ref.current;
    if (!node) return;
    const { observer, entries } = sharedObserver(rootMargin, threshold);
    entries.set(node, { callback: setInView, once });
    observer.observe(node);
    return () => {
      entries.delete(node);
      observer.unobserve(node);
    };
  }, [once, rootMargin, threshold]);

  return { ref, inView };
}

/**
 * Pure spatial placement of the 5 render slabs along the optical Z-axis.
 * At progress=0 the stack sits compressed (0.08 units apart); at progress=1
 * slabs separate smoothly into [-2.10, -1.05, 0, +1.05, +2.10] inspectable planes.
 */
export function computeExplosionOffsets(
  rawProgress: number,
  activeIndex: number,
): PassExplosionOffset[] {
  const t = easeOutExpo(clamp(rawProgress, 0, 1));
  const spacing = lerp(0.08, 1.05, t);
  const centerIndex = (RENDER_PASSES.length - 1) / 2;

  return RENDER_PASSES.map((pass, idx) => {
    const isActive = activeIndex >= 0 && idx === activeIndex;
    const hasFocus = activeIndex >= 0 && t > 0.2;
    const zOffset = (idx - centerIndex) * spacing;
    const liftY = isActive ? lerp(0.04, 0.24, t) : 0;
    const scale = isActive ? lerp(1, 1.045, t) : 1;
    const opacity = !hasFocus ? 1 : isActive ? 1 : lerp(1, 0.72, t);
    return {
      id: pass.id,
      zOffset: Number(zOffset.toFixed(4)),
      liftY: Number(liftY.toFixed(4)),
      scale: Number(scale.toFixed(4)),
      opacity: Number(opacity.toFixed(4)),
    };
  });
}

  targetFps: TargetFps;
  targetMs: number;
  totalMs: number;
  headroomMs: number;
  utilizationPct: number;
  overBudget: boolean;

  totalVramMb: number;

  passes: PassCostBreakdown[];

export const MAX_RESOLUTION_SCALE = 2.0;

/** Computes deterministic GPU frame time, VRAM, draw calls and bottleneck pass. */
export function computePassBudget(
  enabledIds: readonly PassId[],
  rawScale: number,
  targetFps: TargetFps,
): BudgetSummary {
  const scale = Number(clamp(rawScale, MIN_RESOLUTION_SCALE, MAX_RESOLUTION_SCALE).toFixed(2));
  const pixelFactor = scale * scale;
  const enabledSet = new Set(enabledIds);
  const targetMs = targetFps === 120 ? 8.3 : 16.6;

  let totalMs = 0;
  let totalDrawCalls = 0;
  let totalVramMb = 0;
  let bottleneckId: PassId | null = null;
  let maxMs = -1;

  const rawCosts = RENDER_PASSES.map((pass) => {
    const enabled = enabledSet.has(pass.id);
    if (!enabled) {
      return {
        id: pass.id,
        shortName: pass.shortName,
        enabled: false,
        ms: 0,
        drawCalls: 0,
        vramMb: 0,
      };
    }
    const scaledMs = Math.max(0.1, pass.baseMs * (1 + (pixelFactor - 1) * pass.resolutionWeight));
    const ms = Number(scaledMs.toFixed(2));
    const vramMb = Math.round(pass.vramMb * (1 + (pixelFactor - 1) * 0.55));
    totalMs += ms;
    totalDrawCalls += pass.drawCalls;
    totalVramMb += vramMb;
    if (ms > maxMs) {
      maxMs = ms;
      bottleneckId = pass.id;
    }
    return {
      id: pass.id,
      shortName: pass.shortName,
      enabled: true,
      ms,
      drawCalls: pass.drawCalls,
      vramMb,
    };
  });

  const roundedTotal = Number(totalMs.toFixed(1));
  const passes: PassCostBreakdown[] = rawCosts.map((item) => ({
    ...item,
    sharePct: totalMs > 0 ? Math.round((item.ms / totalMs) * 100) : 0,
  }));

  return {
    resolutionScale: scale,
    targetFps,
    targetMs,
    totalMs: roundedTotal,
    headroomMs: Number((targetMs - roundedTotal).toFixed(1)),
    utilizationPct: Math.round((roundedTotal / targetMs) * 100),
    overBudget: roundedTotal > targetMs,
    totalDrawCalls,
    totalVramMb,
    bottleneckId,
    passes,
  };
}

        if (!this.inputs.paused) {
          mesh.rotation.z = this.elapsed * (0.28 + idx * 0.04);
        }
      }
    });

    this.renderer.render(this.scene, this.camera);
  }

  exploded: boolean;
  wireframe: boolean;
  paused: boolean;
  resetTick: number;
};

export const ALL_PASS_IDS: PassId[] = RENDER_PASSES.map((p) => p.id);

export const INITIAL_PIPELINE_STATE: PipelineState = {
  activePassId: "geometry",
  enabledPassIds: [...ALL_PASS_IDS],
  resolutionScale: 1.0,
  targetFps: 60,
  exploded: false,
  wireframe: false,
  paused: false,
  resetTick: 0,
};

export type PipelineAction =
  | { type: "selectPass"; passId: PassId }
  | { type: "stepPass"; delta: -1 | 1 }
  | { type: "togglePassEnabled"; passId: PassId }
  | { type: "setResolutionScale"; scale: number }
  | { type: "setTargetFps"; fps: TargetFps }
  | { type: "toggleExploded" }
  | { type: "setExploded"; exploded: boolean }
  | { type: "toggleWireframe" }
  | { type: "togglePaused" }
  | { type: "reset" };

const VALID_IDS = new Set<string>(ALL_PASS_IDS);

export function pipelineReducer(state: PipelineState, action: PipelineAction): PipelineState {
  switch (action.type) {
    case "selectPass":
      if (!VALID_IDS.has(action.passId)) return state;
      return { ...state, activePassId: action.passId };
    case "stepPass": {
      const idx = ALL_PASS_IDS.indexOf(state.activePassId);
      const next = (idx + action.delta + ALL_PASS_IDS.length) % ALL_PASS_IDS.length;
      return { ...state, activePassId: ALL_PASS_IDS[next] };
    }
    case "togglePassEnabled": {
      if (!VALID_IDS.has(action.passId)) return state;
      const exists = state.enabledPassIds.includes(action.passId);
      const nextIds = exists
        ? state.enabledPassIds.filter((id) => id !== action.passId)
        : ALL_PASS_IDS.filter((id) => id === action.passId || state.enabledPassIds.includes(id));
      return { ...state, enabledPassIds: nextIds };
    }
    case "setResolutionScale": {
      const clamped = Number(clamp(action.scale, MIN_RESOLUTION_SCALE, MAX_RESOLUTION_SCALE).toFixed(2));
      return { ...state, resolutionScale: clamped };
    }
    case "setTargetFps":
      return { ...state, targetFps: action.fps };
    case "toggleExploded":
      return { ...state, exploded: !state.exploded };
    case "setExploded":
      return { ...state, exploded: action.exploded };
    case "toggleWireframe":
      return { ...state, wireframe: !state.wireframe };
    case "togglePaused":
      return { ...state, paused: !state.paused };
    case "reset":
      return { ...INITIAL_PIPELINE_STATE, resetTick: state.resetTick + 1 };
  }
}
