<![CDATA[import { ASSETS } from "./assets.generated";

const BASE = "/templates/cairn";

/** URL for a processed (tritone) image, or null when the owner hasn't generated it yet. */
export function assetSrc(key: string, width: 800 | 1600, format: "webp" | "avif" = "webp"): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

type PictureProps = {
  assetKey: string;
  alt: string;
  className?: string;
  sizes?: string;
  priority?: boolean;
};

/** A responsive <picture>, or a labelled placeholder block until the image exists. */
export function Picture({ assetKey, alt, className, sizes = "(min-width: 900px) 50vw, 100vw", priority }: PictureProps) {
  const webp800 = assetSrc(assetKey, 800);
  if (!webp800) {
    return (
      <div className={`crn-placeholder ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>
        <span className="crn-label">{assetKey}</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  const set = (format: "webp" | "avif") =>
    `${assetSrc(assetKey, 800, format)} 800w, ${assetSrc(assetKey, 1600, format)} 1600w`;
  return (
    <picture className={className}>
      <source type="image/avif" srcSet={set("avif")} sizes={sizes} />
      <img
        src={webp800}
        srcSet={set("webp")}
        sizes={sizes}
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        decoding="async"
        data-asset={assetKey}
      />
    </picture>
  );
}

import { useEffect } from "react";

/** Lenis smooth scrolling, driven by GSAP's ticker and kept in sync with ScrollTrigger. Off under reduced motion. */
export function SmoothScroll() {
  useEffect(() => {
    if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
    let destroy = () => {};
    let cancelled = false;

    Promise.all([import("lenis"), import("gsap"), import("gsap/ScrollTrigger")]).then(
      ([{ default: Lenis }, { gsap }, { ScrollTrigger }]) => {
        if (cancelled) return;
        gsap.registerPlugin(ScrollTrigger);
        const lenis = new Lenis({ lerp: 0.12 });
        lenis.on("scroll", ScrollTrigger.update);
        const tick = (time: number) => lenis.raf(time * 1000);
        gsap.ticker.add(tick);
        gsap.ticker.lagSmoothing(0);
        destroy = () => {
          gsap.ticker.remove(tick);
          lenis.destroy();
        };
      },
    );

    return () => {
      cancelled = true;
      destroy();
    };
  }, []);

  return null;
}

/**
 * A chapter counts as "arrived" when its top reaches the first unobstructed line of the viewport:
 * under the top bar on desktop, under the top bar plus the sticky K2 strip on phones.
 */
function arrivalOffset(stage: HTMLElement | null): number {
  const topbar = parseFloat(getComputedStyle(document.documentElement).getPropertyValue("--topbar")) || 68;
  const stageOnTop = stage && stage.getBoundingClientRect().width >= window.innerWidth - 1;
  return topbar + (stageOnTop ? stage.offsetHeight : 0) + 0.5;

/** Altitude in metres, interpolating camps placed at equal progress spacing (camp i at i/6). */
export function altitudeAt(progress: number): number {
  const p = clamp01(progress) * LAST;
  const i = Math.min(LAST - 1, Math.floor(p));
  const f = p - i;
  const a = CAMPS[i].altitude;
  const b = CAMPS[i + 1].altitude;
  return Math.round(a + (b - a) * f);
}

export function activeCamp(progress: number): number {

/**

 * and each chapter's top. Camp i is reached exactly when chapter i's top crosses the line,
 * so labels line up with chapters even when chapters have different heights.
 */
export function progressFromOffsets(line: number, tops: readonly number[]): number {
  if (tops.length < 2 || line <= tops[0]) return 0;
  const last = tops.length - 1;
  if (line >= tops[last]) return 1;
  let i = 0;
  while (i < last - 1 && line >= tops[i + 1]) i++;
  const f = (line - tops[i]) / (tops[i + 1] - tops[i]);
  return (i + f) / last;

function useScrollThrough<T extends HTMLElement>() {
  const ref = useRef<T>(null);
  const [p, setP] = useState(0);
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    let raf = 0;
    const update = () => {
      raf = 0;
      const r = el.getBoundingClientRect();
      const start = window.innerHeight;
      const end = window.innerHeight * 0.3;
      setP(Math.min(1, Math.max(0, (start - r.top) / (start - end))));
    };
    const onScroll = () => {
      if (!raf) raf = requestAnimationFrame(update);
    };
    update();
    window.addEventListener("scroll", onScroll, { passive: true });
    window.addEventListener("resize", onScroll);
    return () => {

export function FabricZoom({ assetKey, alt, caption }: { assetKey: string; alt: string; caption: string }) {

  const [loupe, setLoupe] = useState<{ x: number; y: number } | null>(null);
  const [tapZoom, setTapZoom] = useState<{ x: number; y: number } | null>(null);

  const local = (e: PointerEvent) => {
    const r = boxRef.current!.getBoundingClientRect();
    return { x: e.clientX - r.left, y: e.clientY - r.top, w: r.width, h: r.height };
  };

  const onMove = (e: PointerEvent) => {

    const track = trackRef.current;
    if (!track) return;
    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    track.scrollBy({ left: dir * track.clientWidth * 0.8, behavior: reduce ? "auto" : "smooth" });
