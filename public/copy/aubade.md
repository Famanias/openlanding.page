<![CDATA[import { ASSETS } from "./assets.generated";

const BASE = "/templates/aubade";

export function assetSrc(key: string, width: 800 | 1600, format: "webp" | "avif" = "webp"): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

type PictureProps = { assetKey: string; alt: string; className?: string; sizes?: string; priority?: boolean };

/** A responsive <picture>, or a labelled placeholder (a quiet colour field) until the image is generated. */
export function Picture({ assetKey, alt, className, sizes = "(min-width: 900px) 40vw, 100vw", priority }: PictureProps) {
  const webp800 = assetSrc(assetKey, 800);
  if (!webp800) {
    return (
      <div className={`au-placeholder ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>
        <span className="au-placeholder__label">{assetKey}.jpg</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  const set = (f: "webp" | "avif") => `${assetSrc(assetKey, 800, f)} 800w, ${assetSrc(assetKey, 1600, f)} 1600w`;
  return (
    <picture className={className}>
      <source type="image/avif" srcSet={set("avif")} sizes={sizes} />
      <img
        src={webp800}
        srcSet={set("webp")}
        sizes={sizes}
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        fetchPriority={priority ? "high" : undefined}
        decoding="async"
        data-asset={assetKey}
      />
    </picture>
  );
}

  { year: 2027, month: 0, label: "January" },
] as const;

const MONTHS_SHORT = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];

const NOTES: Record<string, { note: string; berthsLeft?: number }> = {
  "2026-12-24": { note: "Réveillon dinner on board", berthsLeft: 2 },
  "2026-12-31": { note: "Arrive for Capodanno", berthsLeft: 0 },
};

export function statusFor(berthsLeft: number): DepartureStatus {
  if (berthsLeft <= 0) return "complet";
  if (berthsLeft <= 6) return "few";
  return "open";
}

/** Small deterministic hash so availability looks lived-in but never changes between renders. */
function seeded(id: string): number {
  let h = 2166136261;
  for (let i = 0; i < id.length; i++) {
    h ^= id.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return h >>> 0;
}

function berthsFor(id: string): number {
  const h = seeded(id);
  if (h % 9 === 0) return 0;
  return (h >>> 4) % ROUTE.cabins;
}

/** Every Thursday and Sunday departure in a month (month is 0-based, like Date). */
export function departuresIn(year: number, month: number): Departure[] {
  const out: Departure[] = [];
  const days = new Date(Date.UTC(year, month + 1, 0)).getUTCDate();
  for (let d = 1; d <= days; d++) {
    const dow = new Date(Date.UTC(year, month, d)).getUTCDay();
    if (dow !== 0 && dow !== 4) continue;
    const day = String(d).padStart(2, "0");
    const id = `${year}-${String(month + 1).padStart(2, "0")}-${day}`;
    const special = NOTES[id];
    const berthsLeft = special?.berthsLeft ?? berthsFor(id);
    out.push({
      id,
      weekday: dow === 4 ? "Thu" : "Sun",
      day,
      monthShort: MONTHS_SHORT[month],
      berthsLeft,
      status: statusFor(berthsLeft),
      note: special?.note,
    });
  }
  return out;
}

export function allDepartures(): Departure[] {
  return DEPARTURE_MONTHS.flatMap((m) => departuresIn(m.year, m.month));
}

export function departureLabel(d: Departure): string {
  return `${d.weekday} ${d.day} ${d.monthShort}`;

<![CDATA[import { berthById } from "../content/berths";

export type BookingInput = {
  departure: string;
  berth: string;
  guests: string;
  name: string;
  email: string;
  notes: string;
};

export type BookingField = keyof BookingInput;
export type BookingErrors = Partial<Record<BookingField, string>>;

const EMAIL = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;
export const NOTES_MAX = 280;

/** Client-side rules. `onSale` holds the ids of departures with berths left. */
export function validateBooking(input: BookingInput, onSale: ReadonlySet<string>): BookingErrors {
  const errors: BookingErrors = {};

  if (!input.departure) errors.departure = "Choose a departure.";
  else if (!onSale.has(input.departure)) errors.departure = "That night is full. Choose another departure.";

  const berth = berthById(input.berth);
  if (!berth) errors.berth = "Choose a berth.";

  const guests = Number(input.guests);
  if (!Number.isInteger(guests) || guests < 1 || guests > 2) errors.guests = "One or two guests per cabin.";
  else if (berth && guests > berth.maxGuests) errors.guests = `The ${berth.name} sleeps one.`;

  if (input.name.trim().length < 2) errors.name = "Add the name for the cabin door.";
  if (!EMAIL.test(input.email.trim())) errors.email = "Add an email address for your ticket.";
  if (input.notes.length > NOTES_MAX) errors.notes = `Keep notes to ${NOTES_MAX} characters.`;

  return errors;
}

export function isValid(errors: BookingErrors): boolean {
  return Object.keys(errors).length === 0;
}

/** Per cabin, per night: dinner and breakfast included, a second guest at no extra charge. */
export function fareFor(berthId: string): number {
  return berthById(berthId)?.price ?? 0;
}

function hash(text: string): number {
  let h = 5381;
  for (let i = 0; i < text.length; i++) h = (Math.imul(h, 33) ^ text.charCodeAt(i)) >>> 0;
  return h;
}

export function cabinFor(berthId: string, departure: string, email: string): { voiture: number; cabine: number } {
  const berth = berthById(berthId);
  if (!berth) return { voiture: 0, cabine: 0 };
  return { voiture: berth.voiture, cabine: (hash(`${departure}|${email.trim().toLowerCase()}`) % berth.cabins) + 1 };
}

export function ticketNumber(departure: string, email: string): string {
  const [, mm, dd] = departure.split("-");
  const code = hash(`${email.trim().toLowerCase()}|${departure}`).toString(36).toUpperCase().padStart(4, "0").slice(-4);
  return `AU-${mm}${dd}-${code}`;
}

/** How far the far layer (rate 1) travels across the whole line, in percent of a layer's width. */
export const TRAVEL = 300;
/** Layers are 200% wide and their artwork repeats every 50%. */
export const LAYER_PERIOD = 50;
/** Share of the line over which the tunnel mouth closes (before Brig) or opens (after Domodossola). */

type Offsets = { scrollY: number; top: number; height: number; viewport: number };

/** Scroll position → 0–1 progress through a sticky track. Offsets are cached by the caller. */
export function progressFromOffsets({ scrollY, top, height, viewport }: Offsets): number {
  return progress(scrollY, top, top + Math.max(0, height - viewport));
}

export function legAt(p: number): number {
  const i = LEGS.findIndex((leg) => p < leg.end);
  return i === -1 ? LEGS.length - 1 : Math.max(0, i);
}

export function legProgress(p: number): number {
  const leg = LEGS[legAt(p)];
  return progress(p, leg.start, leg.end);
}

const DEPARTURE = parseClock(STATIONS[0].time);

export function parseClock(time: string): number {
  const [h, m] = time.split(":").map(Number);
  return h * 60 + m;
}

export function formatClock(minutes: number): string {
  const m = Math.round(wrap(minutes, 1440));
  return `${String(Math.floor(m / 60) % 24).padStart(2, "0")}:${String(m % 60).padStart(2, "0")}`;
}

/** Minutes since midnight of the departure day; times before departure belong to the next morning. */
function onTheNight(time: string): number {
  const m = parseClock(time);
  return m < DEPARTURE ? m + 1440 : m;
}

function interpolate(p: number, read: (i: number) => number): number {
  const t = clamp(p, 0, 1);
  const next = STATIONS.findIndex((s) => s.at >= t);
  if (next <= 0) return read(0);
  const a = STATIONS[next - 1];
  const b = STATIONS[next];
  return lerp(read(next - 1), read(next), progress(t, a.at, b.at));
}

export function clockAt(p: number): string {
  return formatClock(interpolate(p, (i) => onTheNight(STATIONS[i].time)));
}

export function kmAt(p: number): number {
  return Math.round(interpolate(p, (i) => STATIONS[i].km));
}

/** Horizontal offset of a landscape layer, in percent of its own width, looping every LAYER_PERIOD. */
export function layerShift(p: number, rate: number): number {
  return wrap(p * rate * TRAVEL, LAYER_PERIOD);
}

const TUNNEL = LEGS.find((leg) => leg.id === "tunnel")!;

/** How closed the tunnel mouth is coming in (enter) and how far it has passed going out (exit), 0–1 each. */
export function tunnelMouth(p: number): { enter: number; exit: number } {
  return {
    enter: progress(p, TUNNEL.start - TUNNEL_RAMP, TUNNEL.start),
    exit: progress(p, TUNNEL.end, TUNNEL.end + TUNNEL_RAMP),
  };
}

/** Keyboard stepping: the next leg's start, or back to the current (then previous) leg's start. */
export function stepTarget(p: number, dir: 1 | -1): number {
  if (dir === 1) return LEGS.find((leg) => leg.start > p + EPS)?.start ?? 1;
  const before = LEGS.filter((leg) => leg.start < p - EPS);
  return before.at(-1)?.start ?? 0;

export function clamp(value: number, min: number, max: number): number {
  return Math.min(max, Math.max(min, value));
}

export function lerp(from: number, to: number, t: number): number {
  return from + (to - from) * t;
}

/** Where `value` sits between `start` and `end`, as 0–1. A zero-width window counts as done. */
export function progress(value: number, start: number, end: number): number {
  if (end <= start) return value >= end ? 1 : 0;
  return clamp((value - start) / (end - start), 0, 1);
}

/** Per-item delay for a cascade, capped so a long list never keeps the last item waiting. */
export function staggerDelay(index: number, step: number, cap: number): number {
  return Math.min(index * step, cap);
}

/** Pointer position inside a card (0–1 on each axis) → tilt in degrees. Top edge tips toward the viewer. */
export function tiltAngles(x: number, y: number, max: number): { rx: number; ry: number } {
  const ry = (clamp(x, 0, 1) - 0.5) * 2 * max;
  const rx = (0.5 - clamp(y, 0, 1)) * 2 * max;
  return { rx: rx + 0, ry: ry + 0 }; // + 0 turns -0 into 0
}

/** Wrap into [0, period), for looping layers. */
export function wrap(value: number, period: number): number {
  return ((value % period) + period) % period;
}

/**
 * Sticky stack: as each card is covered by the next, it scales down and dims. Writes --cover (0–1) per card
 * from one rAF on scroll; layout is pure CSS sticky. Offsets are cached and refreshed on resize.
 */
export function NightStack({ children }: { children: React.ReactNode }) {
  const ref = useRef<HTMLOListElement>(null);

  useEffect(() => {
    const list = ref.current;
    if (!list || document.documentElement.dataset.auMotion !== "live") return;
    const cards = Array.from(list.children) as HTMLElement[];
    let tops: number[] = [];
    let vh = window.innerHeight;
    let raf = 0;

    const measure = () => {
      vh = window.innerHeight;
      // Natural (un-stuck) positions: offsetTop inside the list plus the list's document top.
      const listTop = list.getBoundingClientRect().top + window.scrollY;
      tops = cards.map((c) => listTop + c.offsetTop);
      schedule();
    };
    const frame = () => {
      raf = 0;
      const y = window.scrollY;
      cards.forEach((card, i) => {
        const next = tops[i + 1];
        const cover = next === undefined ? 0 : progress(y + vh, next, next + vh * 0.85);
        card.style.setProperty("--cover", cover.toFixed(3));
      });
    };
    const schedule = () => {
      if (!raf) raf = requestAnimationFrame(frame);

    window.addEventListener("resize", measure, { passive: true });

      cancelAnimationFrame(raf);
      ro.disconnect();
      window.removeEventListener("scroll", schedule);
