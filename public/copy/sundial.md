<![CDATA[import { ASSETS } from "./assets.generated";

const BASE = "/templates/sundial";

export function assetSrc(
  key: string,
  width: 800 | 1600,
  format: "webp" | "avif" = "webp",
): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

type PictureProps = {
  assetKey: string;
  alt: string;
  className?: string;
  sizes?: string;
  priority?: boolean;
};

export function Picture({
  assetKey,
  alt,
  className,
  sizes = "(min-width: 900px) 50vw, 100vw",
  priority,
}: PictureProps) {
  const webp800 = assetSrc(assetKey, 800);
  if (!webp800) {
    return (
      <div className={`sd-placeholder ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>
        <span className="sd-mono">{assetKey}</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  const set = (f: "webp" | "avif") =>
    `${assetSrc(assetKey, 800, f)} 800w, ${assetSrc(assetKey, 1600, f)} 1600w`;
  return (
    <picture className={className}>
      <source type="image/avif" srcSet={set("avif")} sizes={sizes} />
      <img
        src={webp800}
        srcSet={set("webp")}
        sizes={sizes}
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        fetchPriority={priority ? "high" : undefined}
        decoding="async"
        data-asset={assetKey}
      />
    </picture>
  );
}

/** Brew ratio as "1:16.7". */

  const errors: WholesaleErrors = {};

  if (!fields.name.trim()) {
    errors.name = "Tell us who we're talking to.";
  } else if (fields.name.trim().length < 2) {
    errors.name = "That name looks too short.";

  if (!fields.business.trim()) {
    errors.business = "Which business is this for?";
  }

  if (!fields.email.trim()) {
    errors.email = "We need an email to reply to.";
  } else if (!EMAIL_RE.test(fields.email.trim())) {
    errors.email = "That email doesn't look right.";
  }

  if (!VOLUME_OPTIONS.some((o) => o.value === fields.volume)) {
    errors.volume = "Pick a weekly volume.";
  }

  if (fields.message.length > 600) {

export function calculateScrollProgress(
  rectTop: number,
  rectHeight: number,
  windowHeight: number,
): number {
  const travel = rectHeight - windowHeight;
  if (travel <= 0) return 0;

  const scrolled = -rectTop;

    listeners.add(listener);

  reset: () => {
    state = DEFAULT_STATE;
    notify();
  },

 * The roast curve: pure maths for one 12 kg batch, from charge to drop.
 * Everything the stage renders — temperature, stage, development %, the
 * bean's colour — is derived from elapsed seconds through these functions.
 */

/** Total roast time in seconds (11:30). */
export const TOTAL_SECONDS = 690;

/** First crack, in seconds from charge (8:42). Development is measured from here. */
export const CRACK_AT = 522;

/** Charge/drop-in temperature of the drum, °C. */
export const CHARGE_TEMP = 205;

/** Drum temperature at the drop, °C. */
export const DROP_TEMP = 208;

/**
 * Bean-temperature control points [elapsed seconds, °C].
 * Piecewise-linear between knots — the shape of a real logged batch:
 * a plunge to the turning point, then a flattening climb.
 */
const CONTROL_POINTS: ReadonlyArray<readonly [number, number]> = [
  [0, 205],
  [80, 92],
  [180, 146],
  [300, 166],
  [420, 184],
  [522, 196],
  [600, 203],
  [690, 208],
];

export function clampElapsed(elapsed: number): number {
  return Math.max(0, Math.min(TOTAL_SECONDS, elapsed));
}

/** Bean temperature (°C) at a given elapsed time, linearly interpolated. */
export function beanTempAt(elapsed: number): number {
  const t = clampElapsed(elapsed);
  for (let i = 1; i < CONTROL_POINTS.length; i++) {
    const [t1, temp1] = CONTROL_POINTS[i];
    if (t <= t1) {
      const [t0, temp0] = CONTROL_POINTS[i - 1];
      const u = (t - t0) / (t1 - t0);
      return temp0 + (temp1 - temp0) * u;
    }
  }
  return DROP_TEMP;
}

export function developmentPct(elapsed: number): number {
  const t = clampElapsed(elapsed);
  if (t <= CRACK_AT) return 0;
  return ((t - CRACK_AT) / TOTAL_SECONDS) * 100;
}

export type StageMode = "pinned" | "flow";

/**
 * Pinned on wide screens with motion allowed; static flow everywhere else
 * (phones, short screens, prefers-reduced-motion).
 */
export function useStageMode(): StageMode {
  const [mode, setMode] = useState<StageMode>("flow");

  useEffect(() => {
    const mq = window.matchMedia(
      "(min-width: 900px) and (prefers-reduced-motion: no-preference)",
    );

    const update = () => {
      setMode(mq.matches ? "pinned" : "flow");
    };

    update();
    mq.addEventListener("change", update);
    return () => mq.removeEventListener("change", update);
  }, []);

  return mode;
}

/** Adds data-shown when the element scrolls into view. Static under reduced motion. */
export function useReveal<T extends HTMLElement = HTMLDivElement>() {
  const ref = useRef<T>(null);
  const [shown, setShown] = useState(false);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;

    if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
      setShown(true);
      return;
    }

    const observer = new IntersectionObserver(
      (entries) => {
        for (const entry of entries) {
          if (entry.isIntersecting) {
            setShown(true);
            observer.disconnect();
          }
        }
      },
      { rootMargin: "0px 0px -8% 0px", threshold: 0.15 },
    );

    observer.observe(el);
    return () => observer.disconnect();
  }, []);

  return { ref, shown };
}
