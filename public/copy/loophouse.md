<![CDATA[import { ASSETS } from "./assets.generated";

const BASE = "/templates/loophouse";

export function assetSrc(key: string, width: 800 | 1600, format: "webp" | "avif" = "webp"): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

type PictureProps = {
  assetKey: string;
  alt: string;
  className?: string;
  sizes?: string;
  priority?: boolean;
};

export function Picture({
  assetKey,
  alt,
  className,
  sizes = "(min-width: 900px) 50vw, 100vw",
  priority,
}: PictureProps) {
  const webp800 = assetSrc(assetKey, 800);
  if (!webp800) {
    return (
      <div className={`lh-placeholder ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>
        <span className="lh-mono">{assetKey}</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  const set = (f: "webp" | "avif") => `${assetSrc(assetKey, 800, f)} 800w, ${assetSrc(assetKey, 1600, f)} 1600w`;
  return (
    <picture className={className}>
      <source type="image/avif" srcSet={set("avif")} sizes={sizes} />
      <img
        src={webp800}
        srcSet={set("webp")}
        sizes={sizes}
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        fetchPriority={priority ? "high" : undefined}
        decoding="async"
        data-asset={assetKey}
      />
    </picture>
  );
}

  duration = 1200,
  prefix = "",
  suffix = "",
  className = "",

  const { ref, inView } = useInView<HTMLSpanElement>({ once: true });
  const reduced = useReducedMotion();
  const [val, setVal] = useState(0);

  useEffect(() => {
    if (!inView) return;
    if (reduced) {
      setVal(end);
      return;
    }

    const start = performance.now();
    let frameId: number;

    const tick = (now: number) => {
      const elapsed = now - start;
      const progress = Math.min(elapsed / duration, 1);
      // easeOutExpo
      const eased = progress === 1 ? 1 : 1 - Math.pow(2, -10 * progress);
      setVal(eased * end);

      if (progress < 1) {
        frameId = requestAnimationFrame(tick);
      } else {
        setVal(end);
      }
    };

    frameId = requestAnimationFrame(tick);

export function lerp(a: number, b: number, t: number): number {

type InViewOptions = {
  once?: boolean;
  rootMargin?: string;
  threshold?: number | number[];
};

export function useInView<T extends HTMLElement = HTMLDivElement>({
  once = true,
  rootMargin = "0px 0px -10% 0px",
  threshold = 0.1,
}: InViewOptions = {}) {
  const ref = useRef<T | null>(null);
  const [inView, setInView] = useState(false);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;

    if (typeof IntersectionObserver === "undefined") {
      setInView(true);
      return;
    }

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setInView(true);
          if (once) observer.unobserve(el);
        } else if (!once) {
          setInView(false);
        }
      },
      { rootMargin, threshold },
    );

    observer.observe(el);
    return () => observer.disconnect();
  }, [once, rootMargin, threshold]);

  return { ref, inView };
}

export function calculateRequisitionTotals(
  tools: ToolItem[],
  loanDays: number,
  isMember: boolean,
): { count: number; days: number; fee: number; isFreeForMember: boolean } {
  const safeDays = Math.max(1, Math.min(loanDays, 14));
  const count = tools.length;

  if (isMember) {
    return {
      count,
      days: safeDays,
      fee: 0,
      isFreeForMember: true,
    };
  }

  const fee = tools.reduce((sum, tool) => sum + tool.standardDailyFee * safeDays, 0);
  return {
    count,
    days: safeDays,
    fee,
    isFreeForMember: false,
  };
}

export interface WorkbenchState {
  activeCategory: ToolCategory;
  unlatchedToolId: string | null;
  stagedTools: ToolItem[];

}

let state: WorkbenchState = {
  activeCategory: "all",
  unlatchedToolId: "tool-of1400", // Start with the first tool unlatched for immediate discovery
  stagedTools: [TOOLS[0]],
  loanDays: 3,
  isMember: true,
};

const listeners = new Set<() => void>();

function notify() {
  listeners.forEach((l) => l());
}

export const workbenchStore = {
  getSnapshot(): WorkbenchState {
    return state;
  },

  subscribe(listener: () => void): () => void {
    listeners.add(listener);
    return () => listeners.delete(listener);
  },

  setCategory(category: ToolCategory) {
    state = { ...state, activeCategory: category };
    notify();
  },

  unlatchTool(id: string) {
    state = { ...state, unlatchedToolId: state.unlatchedToolId === id ? null : id };
    notify();
  },

  stageTool(tool: ToolItem) {
    if (!state.stagedTools.some((t) => t.id === tool.id)) {
      state = { ...state, stagedTools: [...state.stagedTools, tool], unlatchedToolId: tool.id };
      notify();
    }
  },

  unstageTool(id: string) {
    state = {
      ...state,
      stagedTools: state.stagedTools.filter((t) => t.id !== id),
    };
    notify();
  },

  clearTray() {
    state = { ...state, stagedTools: [] };
    notify();
  },

  setLoanDays(days: number) {
    state = { ...state, loanDays: Math.max(1, Math.min(days, 14)) };
    notify();
  },

  toggleMembership() {
    state = { ...state, isMember: !state.isMember };
    notify();
  },
};

export function useWorkbench(): WorkbenchState {
  return useSyncExternalStore(
    workbenchStore.subscribe,
