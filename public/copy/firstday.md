export function Picture({
  assetKey,
  alt,
  className = "",
  priority = false,
  sizes = "100vw",
}: {
  assetKey: string;
  alt: string;
  className?: string;
  priority?: boolean;
  sizes?: string;
}) {
  const asset = ASSETS[assetKey];
  if (!asset)
    return (
      <div
        className={`fd-placeholder ${className}`}
        role="img"
        aria-label={alt}
      >
        <span>{assetKey}</span>
      </div>
    );
  return (
    <picture className={className}>
      <source
        type="image/avif"
        srcSet={`${assetSrc(assetKey, 800, "avif")} 800w, ${assetSrc(assetKey, 1600, "avif")} 1600w`}
        sizes={sizes}
      />
      <img
        src={assetSrc(assetKey, 1600)!}
        srcSet={`${assetSrc(assetKey, 800)} 800w, ${assetSrc(assetKey, 1600)} 1600w`}
        sizes={sizes}
        width={asset.w}
        height={asset.h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        fetchPriority={priority ? "high" : undefined}
        decoding="async"
      />
    </picture>

export function Reveal({
  children,
  className = "",
  variant = "rise",
  index = 0,
}: {
  children?: React.ReactNode;
  className?: string;
  variant?: "rise" | "mask" | "wipe" | "draw" | "card";
  index?: number;
}) {

    const el = ref.current;
    if (!el) return;
    if (matchMedia("(prefers-reduced-motion:reduce)").matches) {
      el.dataset.in = "true";
      return;

    const target =
      variant === "mask" || variant === "wipe" ? (el.parentElement ?? el) : el;

        observed.delete(target);
        observer?.unobserve(target);
      }
    };
  }, [variant]);
