<![CDATA[import { ASSETS } from "./assets.generated";

const BASE = "/templates/radius";

export function assetSrc(key: string, width: 800 | 1600, format: "webp" | "avif" = "webp"): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

type PictureProps = {
  assetKey: string;
  alt: string;
  className?: string;
  sizes?: string;
  priority?: boolean;
};

export function Picture({ assetKey, alt, className, sizes = "(min-width: 900px) 60vw, 100vw", priority = false }: PictureProps) {
  const src = assetSrc(assetKey, 800);
  if (!src) {
    return <div className={`rd-placeholder ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>{assetKey}</div>;
  }
  const { w, h } = ASSETS[assetKey];
  const srcSet = (format: "webp" | "avif") => `${assetSrc(assetKey, 800, format)} 800w, ${assetSrc(assetKey, 1600, format)} 1600w`;
  return (
    <picture className={className}>
      <source type="image/avif" srcSet={srcSet("avif")} sizes={sizes} />
      <img src={src} srcSet={srcSet("webp")} sizes={sizes} width={w} height={h} alt={alt} loading={priority ? "eager" : "lazy"} fetchPriority={priority ? "high" : undefined} decoding="async" data-asset={assetKey} />
    </picture>
  );
}

<![CDATA[﻿export type Finish = "graphite" | "tangerine" | "cream";
export type Tire = "city" | "gravel";
export type Accessory = "rack" | "basket" | "lights";

export interface BuildSelection {
  finish: Finish;
  tire: Tire;
  accessories: Record<Accessory, boolean>;
}

export type BuildAction =
  | { type: "finish"; finish: Finish }
  | { type: "tire"; tire: Tire }
  | { type: "accessory"; accessory: Accessory; checked: boolean }
  | { type: "reset" };

export const INITIAL_BUILD: BuildSelection = {
  finish: "graphite",
  tire: "city",
  accessories: { rack: false, basket: false, lights: false },
};

const ACCESSORIES: Accessory[] = ["rack", "basket", "lights"];

export function buildReducer(state: BuildSelection, action: BuildAction): BuildSelection {
  switch (action.type) {
    case "finish":
      return { ...state, finish: action.finish };
    case "tire":
      return { ...state, tire: action.tire };
    case "accessory":
      return {
        ...state,
        accessories: { ...state.accessories, [action.accessory]: action.checked },
      };
    case "reset":
      return { ...INITIAL_BUILD, accessories: { ...INITIAL_BUILD.accessories } };
  }
}

export function buildSummary(build: BuildSelection): string {
  const finish = `${build.finish[0].toUpperCase()}${build.finish.slice(1)}`;
  const selected = ACCESSORIES.filter((accessory) => build.accessories[accessory]);
  const accessories = selected.length > 0
    ? `Accessories: ${selected.join(", ")}.`
    : "No accessories selected.";
  return `${finish} finish, ${build.tire} tires. ${accessories}`;
}
