import { CLIPS } from "./videos.generated";

const BASE = "/templates/hiyas";
export function assetSrc(key: string, width: 800 | 1600, format: "webp" | "avif" = "webp"): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

export function Picture({
  assetKey,
  alt,
  className,
  priority = false,
}: {
  assetKey: string;
  alt: string;
  className?: string;
  priority?: boolean;
}) {
  const src = assetSrc(assetKey, 800);
  if (!src) {
    return (
      <div className={`hy-media-slot ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>
        <span>{assetKey}</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  return (
    <picture className={className} data-asset={assetKey}>
      <source
        type="image/avif"
        srcSet={`${assetSrc(assetKey, 800, "avif")} 800w, ${assetSrc(assetKey, 1600, "avif")} 1600w`}
      />
      <img
        src={src}
        srcSet={`${assetSrc(assetKey, 800)} 800w, ${assetSrc(assetKey, 1600)} 1600w`}
        sizes="100vw"
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        fetchPriority={priority ? "high" : undefined}
      />
    </picture>
  );
}

export { CLIPS };

export function VideoPlate({
  clipKey,
  posterKey,
  label,
  className,
  priority = false,
}: {
  clipKey: string;
  posterKey: string;
  label: string;
  className?: string;
  priority?: boolean;
}) {
  return (

        </video>
      )}
    </div>
  );

<![CDATA[export type InquiryInput = { name: string; email: string; peak: string; days: string; support: string; note: string };
export type InquiryErrors = Partial<Record<keyof InquiryInput, string>>;
export type InquiryResult = { ok: true; data: InquiryInput } | { ok: false; errors: InquiryErrors };

const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

export function validateInquiry(input: InquiryInput): InquiryResult {
  const data = Object.fromEntries(Object.entries(input).map(([key, value]) => [key, value.trim()])) as InquiryInput;
  const errors: InquiryErrors = {};
  if (!data.name) errors.name = "Add your name.";
  if (!emailPattern.test(data.email)) errors.email = "Enter a valid email.";
  if (!data.peak) errors.peak = "Choose a peak.";
  const days = Number(data.days);
  if (!Number.isFinite(days) || days < 14 || days > 70) errors.days = "Choose planning days from 14 to 70.";
  if (!["route", "guided", "private"].includes(data.support)) errors.support = "Choose a support mode.";
  if (data.note.length > 600) errors.note = "Keep the note under 600 characters.";
  return Object.keys(errors).length ? { ok: false, errors } : { ok: true, data };
}

export function clampPlannerDays(days: number, min = 14, max = 70): number {
  return Math.round(Math.min(max, Math.max(min, Number.isFinite(days) ? days : min)));

export function routeSegment(progress: number): "approach" | "acclimatize" | "summit" | "return" {
  const p = Math.min(1, Math.max(0, Number.isFinite(progress) ? progress : 0));
  if (p < 0.22) return "approach";
  if (p < 0.68) return "acclimatize";
  if (p < 0.88) return "summit";
  return "return";
}

export function altitudeAt(progress: number, peakElevation: number): number {
  const p = Math.min(1, Math.max(0, Number.isFinite(progress) ? progress : 0));
  const knots = [[0, 0.12], [0.22, 0.28], [0.48, 0.56], [0.72, 0.78], [1, 1]] as const;
  for (let index = 1; index < knots.length; index += 1) {
    const [endProgress, endAltitude] = knots[index];
    const [startProgress, startAltitude] = knots[index - 1];
    if (p <= endProgress) {
      const t = (p - startProgress) / (endProgress - startProgress);
      return Math.round((startAltitude + (endAltitude - startAltitude) * t) * peakElevation);
    }
  }
  return peakElevation;
}

export function campStops(peakElevation: number, days: number, support: SupportLevel): CampStop[] {
  const normalizedDays = clampPlannerDays(days);
  const supportOffset = support === "private" ? -0.02 : support === "route" ? 0.02 : 0;

    { label: "Camp 1", progress: 0.28 },

    { label: "Summit", progress: 1 },
  ];
  return stops.map((stop) => ({ ...stop, altitude: stop.progress === 1 ? peakElevation : altitudeAt(stop.progress, peakElevation) }));
}

}
