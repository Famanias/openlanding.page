<![CDATA[import { ASSETS } from "./assets.generated";
import { CLIPS } from "./videos.generated";

const BASE = "/templates/hiyas";
export function assetSrc(key: string, width: 800 | 1600, format: "webp" | "avif" = "webp"): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

export function Picture({
  assetKey,
  alt,
  className,
  priority = false,
}: {
  assetKey: string;
  alt: string;
  className?: string;
  priority?: boolean;
}) {
  const src = assetSrc(assetKey, 800);
  if (!src) {
    return (
      <div className={`hy-media-slot ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>
        <span>{assetKey}</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  return (
    <picture className={className} data-asset={assetKey}>
      <source
        type="image/avif"
        srcSet={`${assetSrc(assetKey, 800, "avif")} 800w, ${assetSrc(assetKey, 1600, "avif")} 1600w`}
      />
      <img
        src={src}
        srcSet={`${assetSrc(assetKey, 800)} 800w, ${assetSrc(assetKey, 1600)} 1600w`}
        sizes="100vw"
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        fetchPriority={priority ? "high" : undefined}
      />
    </picture>
  );
}

export { CLIPS };

export function VideoPlate({
  clipKey,
  posterKey,
  label,
  className,
  priority = false,
}: {
  clipKey: string;
  posterKey: string;
  label: string;
  className?: string;
  priority?: boolean;
}) {
  return (

          <source src={`${BASE}/clips/${clipKey}.mp4`} type="video/mp4" />
        </video>
      )}
    </div>
  );
}

export function routeSegment(progress: number): "approach" | "acclimatize" | "summit" | "return" {
  const p = Math.min(1, Math.max(0, Number.isFinite(progress) ? progress : 0));
  if (p < 0.22) return "approach";
  if (p < 0.68) return "acclimatize";
  if (p < 0.88) return "summit";
  return "return";
}

export function altitudeAt(progress: number, peakElevation: number): number {
  const p = Math.min(1, Math.max(0, Number.isFinite(progress) ? progress : 0));
  const knots = [[0, 0.12], [0.22, 0.28], [0.48, 0.56], [0.72, 0.78], [1, 1]] as const;
  for (let index = 1; index < knots.length; index += 1) {
    const [endProgress, endAltitude] = knots[index];
    const [startProgress, startAltitude] = knots[index - 1];
    if (p <= endProgress) {
      const t = (p - startProgress) / (endProgress - startProgress);
      return Math.round((startAltitude + (endAltitude - startAltitude) * t) * peakElevation);
    }
  }
  return peakElevation;
}

// Interleave lighter and darker tones so neighboring mountains stay distinct.
// Every relief keeps its original geometry and internal rock/snow contrast.
const SHADES = [1, 0.53, 0.85, 0.45, 0.76, 0.96, 0.6, 0.89, 0.49, 0.8, 0.65, 0.93, 0.56, 0.71];

function toneMatrix(shade: number): string {
  const luminance = [0.2126, 0.7152, 0.0722];
  const blue = [11 / 255, 42 / 255, 74 / 255];
  return [
    ...blue.flatMap((channel) => [
      ...luminance.map((weight) => Number(((1 - channel) * shade * weight).toFixed(5))),
      0,
      channel,
    ]),
    0, 0, 0, 1, 0,
  ].join(" ");
}

function getFallbackPoint(routeStr: string, summit: Point, progress: number): Point {
  const match = routeStr.match(/^M\s*([\d.-]+)\s+([\d.-]+)/);
  const startX = match ? parseFloat(match[1]) : 800;
  const startY = match ? parseFloat(match[2]) : 700;
  return {
    x: Math.round(startX + (summit.x - startX) * progress),
    y: Math.round(startY + (summit.y - startY) * progress),
  };
}
