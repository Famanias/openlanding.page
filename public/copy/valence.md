const BASE = "/templates/valence";

export function assetSrc(
  key: string,
  width: 800 | 1600,
  format: "webp" | "avif" = "webp",
): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

type PictureProps = {
  assetKey: string;
  alt: string;
  className?: string;
  sizes?: string;
  priority?: boolean;
};

export function Picture({
  assetKey,
  alt,
  className,
  sizes = "(min-width: 900px) 50vw, 100vw",
  priority,
}: PictureProps) {
  const webp800 = assetSrc(assetKey, 800);
  if (!webp800) {
    return (
      <div
        className={`vl-placeholder ${className ?? ""}`}
        role="img"
        aria-label={alt}
        data-asset={assetKey}
      >
        <svg
          viewBox="0 0 320 200"
          fill="none"
          xmlns="http://www.w3.org/2000/svg"
          className="w-full h-full object-cover"
          aria-hidden="true"
        >
          <rect width="320" height="200" rx="24" fill="#E8E0F8" />
          <circle cx="96" cy="100" r="46" fill="#D9F2E6" stroke="#1E1B2E" strokeWidth="2" />
          <circle cx="160" cy="100" r="46" fill="#FCE2D5" stroke="#1E1B2E" strokeWidth="2" />
          <circle cx="224" cy="100" r="30" fill="#5B46E8" />
        </svg>
        <span className="vl-mono text-xs">{assetKey}</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  const set = (f: "webp" | "avif") =>
    `${assetSrc(assetKey, 800, f)} 800w, ${assetSrc(assetKey, 1600, f)} 1600w`;
  return (
    <picture className={className}>
      <source type="image/avif" srcSet={set("avif")} sizes={sizes} />
      <img
        src={webp800}
        srcSet={set("webp")}
        sizes={sizes}
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        fetchPriority={priority ? "high" : undefined}
        decoding="async"
        data-asset={assetKey}
      />
    </picture>
  );
}

export interface ValenceStore {
  getSnapshot: () => ValenceState;
  subscribe: (listener: () => void) => () => void;
  setActiveTier: (tier: CaptureTierId) => void;
  setSelectedCohortId: (cohortId: string) => void;
  unlockTier: (tier: CaptureTierId) => void;
  setPreviewFilter: (patch: { companySize?: CompanySizeBand; role?: BuyerRoleId }) => void;
  setActiveChapterId: (chapterId: string) => void;
  recordSqueezeSuccess: (email: string, passCode: string) => void;
  recordLeadCaptureSuccess: (email: string, companySize: CompanySizeBand, role: BuyerRoleId, dossierKey: string) => void;
  toggleAmbientPaused: () => void;
}

export function createValenceStore(initial?: Partial<ValenceState>): ValenceStore {
  let state: ValenceState = {
    activeTier: "splash",
    selectedCohortId: "na-enterprise",
    unlocked: {
      splash: false,
      squeeze: false,
      "lead-gen": false,
    },
    companySize: "201-1000",
    role: "vp-growth",
    activeChapterId: "ch-01",
    squeezeEmail: "",
    squeezePassCode: null,
    dossierKey: null,
    ambientPaused: false,
    ...initial,
  };

  const listeners = new Set<() => void>();
  const emit = () => {
    for (const listener of listeners) listener();
  };

  return {
    getSnapshot: () => state,
    subscribe: (listener) => {
      listeners.add(listener);
      return () => {
        listeners.delete(listener);
      };
    },
    setActiveTier: (tier) => {
      if (state.activeTier === tier) return;
      state = { ...state, activeTier: tier };
      emit();
    },
    setSelectedCohortId: (cohortId) => {
      if (state.selectedCohortId === cohortId) return;
      state = { ...state, selectedCohortId: cohortId };
      emit();
    },
    unlockTier: (tier) => {
      if (state.unlocked[tier]) return;
      state = {
        ...state,
        unlocked: {
          ...state.unlocked,
          [tier]: true,
        },
      };
      emit();
    },
    setPreviewFilter: (patch) => {
      const nextSize = patch.companySize ?? state.companySize;
      const nextRole = patch.role ?? state.role;
      if (nextSize === state.companySize && nextRole === state.role) return;
      state = {
        ...state,
        companySize: nextSize,
        role: nextRole,
      };
      emit();
    },
    setActiveChapterId: (chapterId) => {
      if (state.activeChapterId === chapterId) return;
      state = { ...state, activeChapterId: chapterId };
      emit();
    },
    recordSqueezeSuccess: (email, passCode) => {
      state = {
        ...state,
        activeTier: "squeeze",
        squeezeEmail: email.trim(),
        squeezePassCode: passCode,
        unlocked: {
          ...state.unlocked,
          splash: true,
          squeeze: true,
        },
      };
      emit();
    },
    recordLeadCaptureSuccess: (email, companySize, role, dossierKey) => {
      state = {
        ...state,
        activeTier: "lead-gen",
        squeezeEmail: email.trim(),
        companySize,
        role,
        dossierKey,
        unlocked: {
          splash: true,
          squeeze: true,
          "lead-gen": true,
        },
      };
      emit();
    },
    toggleAmbientPaused: () => {
      state = { ...state, ambientPaused: !state.ambientPaused };
      emit();
    },
  };
}

const EMAIL_PATTERN = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

export interface LeadCaptureInput {
  fullName: string;
  workEmail: string;
  companyName: string;
  companySize: CompanySizeBand;
  role: BuyerRoleId;
}

export type LeadCaptureErrors = Partial<Record<keyof LeadCaptureInput, string>>;

export function validateSqueezeEmail(email: string): string | null {
  const trimmed = email.trim();
  if (!trimmed) {
    return "Enter your work email to unlock the 14-page digest.";
  }
  if (!EMAIL_PATTERN.test(trimmed)) {
    return "Enter a valid email address (e.g. alex@company.com).";
  }
  return null;
}

export function validateLeadCaptureForm(input: LeadCaptureInput): LeadCaptureErrors {
  const errors: LeadCaptureErrors = {};
  const name = input.fullName.trim();
  const email = input.workEmail.trim();
  const company = input.companyName.trim();

  if (name.length < 2) {
    errors.fullName = "Add your full name.";
  }
  if (!email || !EMAIL_PATTERN.test(email)) {
    errors.workEmail = "Enter a valid work email address.";
  }
  if (company.length < 2) {
    errors.companyName = "Add your company or team name.";
  }
  return errors;
}

function deterministicDigits(seed: string): string {
  let hash = 2166136261;
  for (let i = 0; i < seed.length; i += 1) {
    hash ^= seed.charCodeAt(i);
    hash = Math.imul(hash, 16777619);
  }
  const num = Math.abs(hash) % 10000;
  return String(num).padStart(4, "0");
}

export function mintSqueezePassCode(email: string, cohortId: string): string {
  const normalized = `${email.trim().toLowerCase()}::${cohortId.trim().toLowerCase()}`;
  return `SQ-26-${deterministicDigits(normalized)}`;
}

export function mintDossierKey(input: LeadCaptureInput): string {
  const normalized = [
    input.fullName.trim().toLowerCase(),
    input.workEmail.trim().toLowerCase(),
    input.companyName.trim().toLowerCase(),
    input.companySize,
    input.role,
  ].join("::");
  return `VL-2026-${deterministicDigits(normalized)}`;
}

<![CDATA[export function clamp(value: number, min: number, max: number): number {
  if (min > max) return min;
  return Math.min(max, Math.max(min, value));
}

export function lerp(start: number, end: number, t: number): number {
  return start + (end - start) * clamp(t, 0, 1);
}

export function progress(value: number, start: number, end: number): number {
  const span = end - start;
  if (span <= 0) return 0;
  return clamp((value - start) / span, 0, 1);
}

export function staggerDelayMs(index: number, stepMs = 70, maxMs = 600): number {
  return clamp(Math.max(0, index) * stepMs, 0, maxMs);
}

  normX: number,
  normY: number,
  maxDeg = 4,
): { rx: number; ry: number } {
  const x = clamp(normX, 0, 1);

  return {
    rx: Number(((0.5 - y) * 2 * maxDeg).toFixed(2)),
    ry: Number(((x - 0.5) * 2 * maxDeg).toFixed(2)),
  };

  threshold = 0.02,

        for (const entry of entries) {

            setInView(true);
            if (once) observer.unobserve(entry.target);
          } else if (!once) {
            setInView(false);
          }
        }
      },
      { rootMargin, threshold },

    return () => observer.disconnect();
