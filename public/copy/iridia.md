<![CDATA[export interface CommissionFormData {
  fullName: string;
  email: string;
  installationScale: "monumental" | "pavilion" | "gallery" | "digital" | "";
  wavelengthBand: "solar" | "auroral" | "laser" | "broadband" | "";
  targetDate: string;
  notes?: string;
}

export interface ValidationResult {
  isValid: boolean;
  errors: Partial<Record<keyof CommissionFormData, string>>;
}

const EMAIL_REGEX = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

export function validateCommissionForm(data: CommissionFormData): ValidationResult {
  const errors: Partial<Record<keyof CommissionFormData, string>> = {};

  if (!data.fullName || data.fullName.trim().length < 2) {
    errors.fullName = "Please enter your name (minimum 2 characters).";
  }

  if (!data.email || !EMAIL_REGEX.test(data.email.trim())) {
    errors.email = "Please enter a valid architectural or studio email address.";
  }

  if (!data.installationScale) {
    errors.installationScale = "Please select an installation scale tier.";
  }

  if (!data.wavelengthBand) {
    errors.wavelengthBand = "Please select a target optical wavelength band.";
  }

  if (!data.targetDate) {
    errors.targetDate = "Please select a proposed commission date.";
  } else {
    const selected = new Date(data.targetDate);
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    if (isNaN(selected.getTime()) || selected < today) {
      errors.targetDate = "Commission date must be in the future.";
    }
  }

  if (data.notes && data.notes.length > 500) {
    errors.notes = "Notes must not exceed 500 characters.";
  }

  return {
    isValid: Object.keys(errors).length === 0,
    errors,
  };
}

precision highp float;

out vec4 fragColor;

uniform vec2 u_resolution;

uniform vec3 u_stop1;
uniform vec3 u_stop2;

           -0.577350269189626, 0.024390243902439);

  vec2 x0 = v -   i + dot(i, C.xx);

  m = m*m ;

/**
 * Calculates refractive index at a given wavelength using Cauchy's empirical formula:
 * n(λ) = A + B / λ² (where λ is in micrometers)
 */
export function cauchyRefraction(a: number, b: number, wavelengthNm: number): number {
  if (wavelengthNm <= 0) return a;
  const lambdaMicrons = wavelengthNm / 1000;
  return a + b / (lambdaMicrons * lambdaMicrons);
}

/**
 * Calculates Abbe number (constringence Vd), measuring optical dispersion:
 * Vd = (nD - 1) / (nF - nC)
 * Fraunhofer lines:
 * nD = 589.3 nm (yellow sodium)
 * nF = 486.1 nm (blue hydrogen)
 * nC = 656.3 nm (red hydrogen)
 */
export function calculateAbbeNumber(nD: number, nF: number, nC: number): number {
  const dispersion = nF - nC;
  if (Math.abs(dispersion) < 0.0001) return 100.0;
  return (nD - 1.0) / dispersion;
}

/**
 * Maps physical wavelength in nanometers (380 to 700 nm) to approximate Hue angle (0 to 360 deg).
 */
export function wavelengthToHue(wavelengthNm: number): number {
  const clamped = Math.max(380, Math.min(700, wavelengthNm));
  // 380 nm -> Violet (280°), 480 nm -> Cyan (180°), 530 nm -> Green (130°), 590 nm -> Yellow (50°), 650 nm -> Red (0°)
  if (clamped < 440) {
    return 280 - ((clamped - 380) / 60) * 40; // 280 -> 240
  } else if (clamped < 490) {
    return 240 - ((clamped - 440) / 50) * 60; // 240 -> 180 (Cyan)
  } else if (clamped < 550) {
    return 180 - ((clamped - 490) / 60) * 60; // 180 -> 120 (Green)
  } else if (clamped < 600) {
    return 120 - ((clamped - 550) / 50) * 70; // 120 -> 50 (Yellow)
  } else {
    return Math.max(0, 50 - ((clamped - 600) / 100) * 50); // 50 -> 0 (Red)
  }
}

/**
 * Simulates optical radiative energy flux in lm/m² given refractive index and wavelength.
 */
export function calculateFlux(refractiveIndex: number, wavelengthNm: number, baseLumens = 1000): number {
  // Higher refraction and shorter wavelengths concentrate flux density
  const wavelengthFactor = 550 / Math.max(380, Math.min(750, wavelengthNm));
  const indexFactor = Math.pow(refractiveIndex, 1.4);
  return Math.round(baseLumens * wavelengthFactor * indexFactor);
}

import { calculateCameraOrbit, type CameraOrbitState } from "./physics";

export interface SpectralStoreState {
  activePresetId: SpectralPresetId;
  dispersion: number;
  turbulence: number;
  flowVelocity: number;
  chromaticSpread: number;
  scrollProgress: number;
  pointer: { x: number; y: number; vx: number; vy: number };

  prismYaw: number;
  prismPitch: number;
  version: number;
}

type Listener = () => void;

export class SpectralStore {
  private state: SpectralStoreState;
  private listeners: Set<Listener> = new Set();

  constructor(initialPreset: SpectralPresetId = "solar-flare") {
    const preset = SPECTRAL_PRESETS[initialPreset];
    this.state = {
      activePresetId: initialPreset,
      dispersion: preset.refractiveIndex,
      turbulence: preset.turbulence,
      flowVelocity: preset.flowVelocity,
      chromaticSpread: preset.chromaticSpread,
      scrollProgress: 0,
      pointer: { x: 0, y: 0, vx: 0, vy: 0 },
      cameraOrbit: calculateCameraOrbit(0, 0, 0),
      reducedMotion: false,
      prismYaw: -0.28,
      prismPitch: 0.1,
      version: 0,
    };
  }

  public getSnapshot = (): SpectralStoreState => {
    return this.state;
  };

  public subscribe = (listener: Listener): (() => void) => {
    this.listeners.add(listener);
    return () => {
      this.listeners.delete(listener);
    };
  };

  private notify() {
    this.state = { ...this.state, version: this.state.version + 1 };
    for (const listener of this.listeners) {
      listener();
    }
  }

  public setPreset(id: SpectralPresetId) {
    if (!SPECTRAL_PRESETS[id]) return;
    const preset = SPECTRAL_PRESETS[id];
    this.state = {
      ...this.state,
      activePresetId: id,
      dispersion: preset.refractiveIndex,
      turbulence: preset.turbulence,
      flowVelocity: preset.flowVelocity,
      chromaticSpread: preset.chromaticSpread,
    };
    this.notify();
  }

  public setParameters(params: Partial<{
    dispersion: number;
    turbulence: number;
    flowVelocity: number;
    chromaticSpread: number;
  }>) {
    this.state = {
      ...this.state,
      dispersion: params.dispersion ?? this.state.dispersion,
      turbulence: params.turbulence ?? this.state.turbulence,
      flowVelocity: params.flowVelocity ?? this.state.flowVelocity,
      chromaticSpread: params.chromaticSpread ?? this.state.chromaticSpread,
    };
    this.notify();
  }

  public setPrismOrientation(yaw: number, pitch: number) {
    if (Math.abs(this.state.prismYaw - yaw) < 0.001 && Math.abs(this.state.prismPitch - pitch) < 0.001) return;
    this.state = {
      ...this.state,
      prismYaw: yaw,
      prismPitch: pitch,
    };
    this.notify();
  }

  public setScrollProgress(progress: number) {
    const clamped = Math.max(0, Math.min(1, progress));
    if (Math.abs(clamped - this.state.scrollProgress) < 0.001) return;
    const orbit = this.state.reducedMotion
      ? { pitchDeg: 0, yawDeg: 0, rollDeg: 0, translationZ: 0, focalDepth: 1.5 }
      : calculateCameraOrbit(clamped, this.state.pointer.x, this.state.pointer.y);

    this.state = {
      ...this.state,
      scrollProgress: clamped,
      cameraOrbit: orbit,
    };
    this.notify();
  }

  public setPointer(x: number, y: number, vx = 0, vy = 0) {
    const orbit = this.state.reducedMotion
      ? { pitchDeg: 0, yawDeg: 0, rollDeg: 0, translationZ: 0, focalDepth: 1.5 }
      : calculateCameraOrbit(this.state.scrollProgress, x, y);

    this.state = {
      ...this.state,
      pointer: { x, y, vx, vy },
      cameraOrbit: orbit,
    };
    this.notify();
  }

  public setReducedMotion(reduced: boolean) {
    if (this.state.reducedMotion === reduced) return;
    const orbit = reduced
      ? { pitchDeg: 0, yawDeg: 0, rollDeg: 0, translationZ: 0, focalDepth: 1.5 }
      : calculateCameraOrbit(this.state.scrollProgress, this.state.pointer.x, this.state.pointer.y);

    this.state = {
      ...this.state,
      reducedMotion: reduced,
      cameraOrbit: orbit,
    };
    this.notify();
  }

  public getActivePreset(): SpectralPreset {
    return SPECTRAL_PRESETS[this.state.activePresetId];
  }
}
