<![CDATA[import { ASSETS } from "./assets.generated";

const BASE = "/templates/orrery";

/** URL for a processed image, or null when the owner hasn't generated it yet. */
export function assetSrc(key: string, width: 800 | 1600, format: "webp" | "avif" = "webp"): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

type PictureProps = {
  assetKey: string;
  alt: string;
  className?: string;
  sizes?: string;
  priority?: boolean;
};

/** A responsive <picture>, or a labelled placeholder block until the image exists. */
export function Picture({ assetKey, alt, className, sizes = "(min-width: 900px) 50vw, 100vw", priority }: PictureProps) {
  const webp800 = assetSrc(assetKey, 800);
  if (!webp800) {
    return (
      <div className={`orr-placeholder ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>
        <span className="orr-label">{assetKey}</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  const set = (format: "webp" | "avif") => `${assetSrc(assetKey, 800, format)} 800w, ${assetSrc(assetKey, 1600, format)} 1600w`;
  return (
    <picture className={className}>
      <source type="image/avif" srcSet={set("avif")} sizes={sizes} />
      <img
        src={webp800}
        srcSet={set("webp")}
        sizes={sizes}
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        decoding="async"
        data-asset={assetKey}
      />
    </picture>
  );
}

<![CDATA[export type DeckState = { index: number; count: number };

export type DeckAction =
  | { type: "next" | "prev" | "first" | "last" }
  | { type: "goto"; index: number };

const clamp = (index: number, count: number) => Math.min(count - 1, Math.max(0, index));

export function deckReducer(state: DeckState, action: DeckAction): DeckState {
  const target = (() => {
    switch (action.type) {
      case "next":
        return state.index + 1;
      case "prev":
        return state.index - 1;
      case "first":
        return 0;
      case "last":
        return state.count - 1;
      case "goto":
        return action.index;
    }
  })();
  const index = clamp(target, state.count);
  return index === state.index ? state : { ...state, index };
}

export function formatCounter(index: number, count: number): string {
  const pad = (n: number) => String(n).padStart(2, "0");
  return `${pad(index + 1)} / ${pad(count)}`;
}

/** Advances the build every 450 ms while `active`, like a keynote build. Fully built at once under reduced motion. */
export function useOrbitBuild(active: boolean): OrbitStep {
  const reduce = useReducedMotion();
  const [step, setStep] = useState<OrbitStep>(0);

  useEffect(() => {
    if (reduce) return;
    if (!active) {
      setStep(0);
      return;
    }
    const id = window.setInterval(() => {
      setStep((s) => (s >= FINAL ? s : ((s + 1) as OrbitStep)));
    }, STEP_MS);
    return () => window.clearInterval(id);
  }, [active, reduce]);

  return reduce ? FINAL : step;
}

const COUNT_MS = 1200;

/** Counts from 0 to `target` with a strong ease-out while `active`; static under reduced motion. */
function useCountUp(target: number, active: boolean) {
  const reduce = useReducedMotion();
  const [value, setValue] = useState(target);

  useEffect(() => {
    if (reduce || !active) {
      setValue(target);
      return;
    }
    let raf = 0;
    const start = performance.now();
    const tick = (now: number) => {
      const t = Math.min(1, (now - start) / COUNT_MS);
      const eased = 1 - Math.pow(1 - t, 4);
      setValue(target * eased);
      if (t < 1) raf = requestAnimationFrame(tick);
    };
    setValue(0);
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [active, reduce, target]);

  return value;
}
