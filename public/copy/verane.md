/** URL for a processed image, or null when not yet generated. */
export function assetSrc(
  key: string,
  width: 800 | 1600,
  format: "webp" | "avif" = "webp",
): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

export type PictureProps = {
  assetKey: string;
  alt: string;
  className?: string;
  sizes?: string;
  priority?: boolean;
  aspectRatio?: string;
  dark?: boolean;
};

/**
 * A responsive <picture>, or a styled placeholder block labeled with its assetKey.
 */
export function Picture({
  assetKey,
  alt,
  className,
  sizes = "(min-width: 1024px) 50vw, 100vw",
  priority,
  aspectRatio,
  dark,
}: PictureProps) {
  const webp800 = assetSrc(assetKey, 800);
  if (!webp800) {
    return (
      <div
        className={`vr-placeholder ${dark ? "vr-placeholder--dark" : ""} ${className ?? ""}`}
        style={aspectRatio ? { aspectRatio } : undefined}
        role="img"

      >
        <span className="vr-mono">{assetKey}</span>
        <span style={{ fontSize: "0.8rem", opacity: 0.75 }}>{alt}</span>
      </div>
    );
  }

    `${assetSrc(assetKey, 800, format)} 800w, ${assetSrc(assetKey, 1600, format)} 1600w`;

interface LogoMarkProps extends SVGProps<SVGSVGElement> {
  size?: number;
  topLevel?: number; // 0 to 1
  heartLevel?: number; // 0 to 1
  baseLevel?: number; // 0 to 1
}

/**
 * Maison Vérane Mark: A stylized perfume drop divided into the three evaporation tiers:
 * Top (evaporates early), Heart (mid layer), and Base (resins/woods that hold to skin).
 * When levels are passed (0..1), the liquid bands scale/drain with the shared dry-down clock.
 */
export function LogoMark({
  size = 32,
  topLevel = 0,
  heartLevel = 0.55,
  baseLevel = 1,
  className,
  ...props
}: LogoMarkProps) {
  // Height allocation in the 64x64 viewBox:
  // Top band: y=20..33 (h=13)
  // Heart band: y=33..45 (h=12)
  // Base band: y=45..60 (h=15)
  return (
    <svg
      viewBox="0 0 64 64"
      width={size}
      height={size}
      role="img"
      aria-label="Maison Vérane drop mark"
      className={className}
      {...props}
    >
      <defs>
        <clipPath id="vr-drop-clip">
          <path d="M32 5C26 14 13 28 13 40a19 19 0 0 0 38 0C51 28 38 14 32 5Z" />
        </clipPath>
      </defs>

      {/* Internal liquid layers, clipped inside the flacon drop */}
      <g clipPath="url(#vr-drop-clip)">

export interface ClockStore {
  getState: () => ClockState;
  setHour: (hour: number) => void;
  setFragranceSlug: (slug: string) => void;
  subscribe: (listener: () => void) => () => void;
}

const DEFAULT_STATE: ClockState = {
  hour: 0,
  fragranceSlug: "iris-calcaire",
};

/** Creates an observable clock store compatible with React's useSyncExternalStore. */
export function createClockStore(initial?: Partial<ClockState>): ClockStore {
  let state: ClockState = {
    hour: initial?.hour ?? DEFAULT_STATE.hour,
    fragranceSlug: initial?.fragranceSlug ?? DEFAULT_STATE.fragranceSlug,
  };

  const listeners = new Set<() => void>();

  function notify() {
    for (const listener of listeners) {
      listener();
    }
  }

  return {
    getState: () => state,
    setHour: (nextHour: number) => {
      const clamped = Math.max(0, Math.min(12, Math.round(nextHour * 100) / 100));
      if (clamped === state.hour) return;
      state = { ...state, hour: clamped };
      notify();
    },
    setFragranceSlug: (nextSlug: string) => {
      if (nextSlug === state.fragranceSlug) return;
      state = { ...state, fragranceSlug: nextSlug };
      notify();
    },
    subscribe: (listener: () => void) => {
      listeners.add(listener);
      return () => {
        listeners.delete(listener);
      };
    },
  };
}

  | "next15"
  | "prev15"

  | "prevHour"

/** Format a decimal hour (0..12) into "H HH:MM" format. */
export function formatHour(hour: number): string {
  const clamped = Math.max(0, Math.min(12, hour));
  const totalMinutes = Math.round(clamped * 60);
  const h = Math.floor(totalMinutes / 60);
  const m = totalMinutes % 60;
  const padH = h.toString().padStart(2, "0");
  const padM = m.toString().padStart(2, "0");
  return `H ${padH}:${padM}`;
}

/**
 * Progress (0..1) -> Hour (0..12)
 * Quadratic curve: hour = 12 * progress^2
 * Gives top notes (first 45 min) the first 25% of scroll travel.
 */
export function progressToHour(progress: number): number {
  const p = Math.max(0, Math.min(1, progress));
  return 12 * (p * p);
}

/**
 * Hour (0..12) -> Progress (0..1)
 * Inverted quadratic: progress = sqrt(hour / 12)
 */
export function hourToProgress(hour: number): number {
  const h = Math.max(0, Math.min(12, hour));
  return Math.sqrt(h / 12);
}

/** Compute updated hour when stepped via keyboard / buttons. */
export function stepHour(currentHour: number, step: ClockStep): number {
  let next = currentHour;
  switch (step) {
    case "next15":
      next = currentHour + 0.25;
      break;
    case "prev15":
      next = currentHour - 0.25;
      break;
    case "nextHour":
      next = currentHour + 1;
      break;
    case "prevHour":
      next = currentHour - 1;
      break;
    case "start":
      next = 0;
      break;
    case "end":
      next = 12;
      break;
  }
  return Math.max(0, Math.min(12, Math.round(next * 100) / 100));

  onset: number; // hour fully present (e.g. 0, 0.3)
  fade: number; // hour starts to dissipate
  gone: number; // hour completely evaporated
  description: string;
}

export interface NoteCalculated {
  intensity: number; // 0..1
  remaining: number; // 0..1
  state: NoteState;
}

export interface TierLevels {
  top: number;
  heart: number;
  base: number;
}

/** Calculate the instant intensity, remaining volume, and state of a note at a given hour. */
export function calculateNoteState(note: Note, hour: number): NoteCalculated {
  const h = Math.max(0, hour);
  const { onset, fade, gone } = note;

  // Remaining (in flacon/on skin): 1 until fade, linearly 0 at gone
  let remaining = 1;
  if (h <= fade) {
    remaining = 1;
  } else if (h >= gone) {
    remaining = 0;
  } else {
    remaining = Math.max(0, Math.min(1, 1 - (h - fade) / (gone - fade)));
  }

  // Intensity (olfactory prominence):
  let intensity = 0;
  let state: NoteState = "present";

  if (h >= gone) {
    intensity = 0;
    state = "gone";
  } else if (h >= fade) {
    intensity = remaining;
    state = "fading";
  } else if (h < onset) {
    state = "emerging";
    const rampDuration = Math.min(onset, 0.5);
    if (rampDuration > 0 && h >= onset - rampDuration) {
      intensity = (h - (onset - rampDuration)) / rampDuration;
    } else {
      intensity = 0;
    }
  } else {
    intensity = 1;
    state = "present";
  }

  return {
    intensity: Math.max(0, Math.min(1, intensity)),
    remaining: Math.max(0, Math.min(1, remaining)),
    state,
  };
}

/** Calculate the liquid levels across the three flacon tiers (top, heart, base). */
export function calculateTierLevels(notes: Note[], hour: number): TierLevels {
  const levels: TierLevels = { top: 0, heart: 0, base: 0 };

  for (const tier of ["top", "heart", "base"] as const) {
    const tierNotes = notes.filter((n) => n.tier === tier);
    if (tierNotes.length === 0) {
      levels[tier] = 0;
    } else {
      let maxRemaining = 0;
      for (const n of tierNotes) {
        const { remaining } = calculateNoteState(n, hour);
        if (remaining > maxRemaining) maxRemaining = remaining;
      }
      levels[tier] = maxRemaining;
    }
  }

  return levels;
}

/** Returns the dominant tier at the given hour. */
export function getDominantTier(notes: Note[], hour: number): OlfactoryTier {
  const intensities: Record<OlfactoryTier, number> = { top: 0, heart: 0, base: 0 };

  for (const n of notes) {
    const { intensity } = calculateNoteState(n, hour);
    intensities[n.tier] += intensity;
  }

  // Determine highest with tie-breaking top -> heart -> base
  let bestTier: OlfactoryTier = "top";
  let maxScore = intensities.top;

  if (intensities.heart > maxScore) {
    bestTier = "heart";
    maxScore = intensities.heart;
  }
  if (intensities.base > maxScore) {
    bestTier = "base";
  }

  return bestTier;
}

 * Calculate progress (0..1) through a pinned section using its bounding client rect.
 *
 * @param rectTop - getBoundingClientRect().top of the outer section
 * @param rectHeight - offsetHeight or getBoundingClientRect().height of the outer section
 * @param windowHeight - window.innerHeight
 */
export function calculateScrollProgress(
  rectTop: number,
  rectHeight: number,
  windowHeight: number,
): number {
  const travel = rectHeight - windowHeight;
  if (travel <= 0) return 0;

  const scrolled = -rectTop;

  return Math.max(0, Math.min(1, progress));

/** Compute derived blend composition based on the 3 chosen vials. */
export function deriveBespokeBlend(
  topId: string,
  heartId: string,
  baseId: string,
): BespokeBlend {
  const topVial = getOrganVial("top", topId);
  const heartVial = getOrganVial("heart", heartId);
  const baseVial = getOrganVial("base", baseId);

  const topIdx = ORGAN_SHELVES.top.findIndex((v) => v.id === topVial.id) + 1;
  const heartIdx = ORGAN_SHELVES.heart.findIndex((v) => v.id === heartVial.id) + 1;
  const baseIdx = ORGAN_SHELVES.base.findIndex((v) => v.id === baseVial.id) + 1;

  const batchCode = `B-${topIdx.toString().padStart(2, "0")}${heartIdx}${baseIdx}`;
  const title = `${topVial.shortName} · ${heartVial.shortName} · ${baseVial.shortName}`;

  // Longevity estimation
  const maxGone = Math.max(topVial.note.gone, heartVial.note.gone, baseVial.note.gone);
  const longevity = `Lasts past ${Math.floor(maxGone)} h on skin`;
  const formula = `${topVial.origin} + ${heartVial.origin} + ${baseVial.origin}`;

  return {
    title,
    batchCode,
    notes: [topVial.note, heartVial.note, baseVial.note],
    longevity,
    formula,
  };
}

export function DryDownSection() {
  const outerRef = useRef<HTMLDivElement>(null);
  const { setHour } = useClock();
  const mode = useStageMode();

  useEffect(() => {
    if (mode !== "pinned") return;

    let ticking = false;

    const handleScroll = () => {
      if (!ticking) {
        window.requestAnimationFrame(() => {
          if (!outerRef.current) return;
          const rect = outerRef.current.getBoundingClientRect();
          const winH = window.innerHeight;

          // Only drive clock when section is actively within the pinned scroll envelope
          if (rect.top <= 0 && rect.bottom >= winH) {
            const progress = calculateScrollProgress(rect.top, rect.height, winH);
            const hour = progressToHour(progress);
            setHour(hour);
          }
          ticking = false;
        });

      }
    };

    window.addEventListener("scroll", handleScroll, { passive: true });
    return () => window.removeEventListener("scroll", handleScroll);
  }, [mode, setHour]);

    <section
      id="drydown"
      ref={outerRef}
      data-mode={mode}
      className="vr-drydown-outer vr-dark"
