<![CDATA[import { ASSETS } from "./assets.generated";

const BASE = "/templates/wowflutter";

/** URL for a processed (ink-on-stock) image, or null when the owner hasn't generated it yet. */
export function assetSrc(key: string, width: 800 | 1600, format: "webp" | "avif" = "webp"): string | null {
  return key in ASSETS ? `${BASE}/${key}-${width}.${format}` : null;
}

type PictureProps = {
  assetKey: string;
  alt: string;
  className?: string;
  sizes?: string;
  priority?: boolean;
};

/** A responsive <picture>, or a labelled placeholder block until the image exists. */
export function Picture({ assetKey, alt, className, sizes = "(min-width: 1000px) 40vw, 90vw", priority }: PictureProps) {
  const webp800 = assetSrc(assetKey, 800);
  if (!webp800) {
    return (
      <div className={`wf-placeholder ${className ?? ""}`} role="img" aria-label={alt} data-asset={assetKey}>
        <span className="wf-label">{assetKey}</span>
      </div>
    );
  }
  const { w, h } = ASSETS[assetKey];
  const set = (format: "webp" | "avif") =>
    `${assetSrc(assetKey, 800, format)} 800w, ${assetSrc(assetKey, 1600, format)} 1600w`;
  return (
    <picture className={className}>
      <source type="image/avif" srcSet={set("avif")} sizes={sizes} />
      <img
        src={webp800}
        srcSet={set("webp")}
        sizes={sizes}
        width={w}
        height={h}
        alt={alt}
        loading={priority ? "eager" : "lazy"}
        fetchPriority={priority ? "high" : undefined}
        decoding="async"
        data-asset={assetKey}
      />
    </picture>
  );
}

/** A picture on a printed plate: the image sits on stock with a pink pass printed a few millimetres off-register. */
export function Plate({ className, ...picture }: PictureProps) {
  return (
    <figure className={`wf-plate ${className ?? ""}`}>
      <Picture {...picture} />
    </figure>
  );
}

/** Releasing speed in px/ms toward finishing the turn, from the last ~90 ms of pointer samples. */
function releaseVelocity(samples: Drag["samples"], dir: Direction): number {
  const last = samples[samples.length - 1];
  const first = samples.find((s) => last.t - s.t <= 90) ?? last;
  const dt = last.t - first.t;
  if (dt <= 0) return 0;
  const vx = (last.x - first.x) / dt;
  return dir === 1 ? -vx : vx;
}

  spread: number;
  /** The leaf currently in the air, if any. `progress` is 0 untouched .. 1 finished. */
  turning: { leaf: number; dir: Direction; progress: number } | null;
  /** Continuous position through the book (3.25 = a quarter of the way from spread 3 to 4). */
  position: number;
};

export type EngineHooks = {
  render(view: EngineView): void;
  /** A turn ended (or a jump happened): `spread` is now the resting spread. */
  commit(spread: number): void;
  request(callback: (now: number) => void): number;
  cancel(id: number): void;
  now(): number;
};

type Turning = { leaf: number; dir: Direction; p: number; dragging: boolean };

/**
 * The page-turn state machine. It owns which leaf is in the air and how far it has gone, and talks to the
 * outside only through `hooks`, so it can be driven by a fake clock in tests. One leaf moves at a time;
 * progress is the single source of truth, so a turn can be grabbed, reversed or redirected at any frame.
 */
export class TurnEngine {
  private _spread = 0;
  private goal = 0;
  private turning: Turning | null = null;
  private raf = 0;
  private last = 0;
  private reduced = false;

  constructor(private readonly hooks: EngineHooks) {}

  get spread(): number {
    return this._spread;
  }

  setReduced(reduced: boolean): void {
    this.reduced = reduced;
  }

  /** Instant move with no animation: first paint from a hash, reduced motion, resets. */
  jumpTo(spread: number): void {
    this.stopLoop();
    this.turning = null;
    this._spread = this.goal = clampSpread(spread);
    this.emit();
    this.hooks.commit(this._spread);
  }

  /** Animate to a spread, turning one leaf at a time. */
  goTo(spread: number): void {
    if (this.turning?.dragging) return;
    const target = clampSpread(spread);
    if (this.reduced) return this.jumpTo(target);
    this.goal = target;
    this.schedule();
  }

  /** One page forward (1) or back (-1) from wherever the queue currently ends. */
  step(dir: Direction): void {
    this.goTo(this.goal + dir);
  }

  /**
   * The pointer takes hold of a leaf. Returns the direction and progress to continue from, or null when
   * there is nothing to turn that way. A leaf already in the air is taken over where it is.
   */
  grab(dir: Direction): { dir: Direction; progress: number } | null {
    if (this.reduced) return null;
    this.stopLoop();
    if (this.turning) {
      this.turning.dragging = true;
      this.goal = this._spread;
      return { dir: this.turning.dir, progress: this.turning.p };
    }
    if (dir === 1 ? this._spread >= LEAF_COUNT : this._spread <= 0) return null;
    this.goal = this._spread;
    this.turning = { leaf: dir === 1 ? this._spread : this._spread - 1, dir, p: 0, dragging: true };
    return { dir, progress: 0 };
  }

  drag(progress: number): void {
    if (!this.turning?.dragging) return;
    this.turning.p = Math.min(1, Math.max(0, progress));
    this.emit();
  }

  /** The pointer lets go. `velocity` is px/ms toward finishing the turn. */
  release(velocity: number): void {
    const t = this.turning;
    if (!t?.dragging) return;
    t.dragging = false;
    this.goal = settleTarget(t.p, velocity) === 1 ? this._spread + t.dir : this._spread;
    this.last = this.hooks.now();
    this.schedule();
  }

  dispose(): void {
    this.stopLoop();
  }

  private view(): EngineView {
    const t = this.turning;
    return {
      spread: this._spread,
      turning: t ? { leaf: t.leaf, dir: t.dir, progress: t.p } : null,
      position: t ? bookPosition(this._spread, t.dir, t.p) : this._spread,
    };
  }

  private emit(): void {
    this.hooks.render(this.view());
  }

  private schedule(): void {
    if (this.raf) return;
    this.last = this.hooks.now();
    this.raf = this.hooks.request(this.frame);
  }

  private stopLoop(): void {
    if (this.raf) this.hooks.cancel(this.raf);
    this.raf = 0;
  }

  private frame = (now: number): void => {
    this.raf = 0;
    const dt = Math.min(48, Math.max(0, now - this.last));
    this.last = now;

    if (!this.turning) {
      if (this.goal === this._spread) return;
      const dir: Direction = this.goal > this._spread ? 1 : -1;
      this.turning = { leaf: dir === 1 ? this._spread : this._spread - 1, dir, p: 0, dragging: false };
    }

    const t = this.turning;
    if (!t.dragging) {
      const finishing = t.dir === 1 ? this.goal > this._spread : this.goal < this._spread;
      const target: 0 | 1 = finishing ? 1 : 0;
      // Long jumps run faster so a contents link doesn't take seven full turns.
      const speed = Math.abs(this.goal - this._spread) > 1 ? 2.4 : 1;
      t.p = stepProgress(t.p, target, dt * speed);
      if (t.p === target) {
        if (target === 1) this._spread += t.dir;
        this.turning = null;
        this.emit();
        this.hooks.commit(this._spread);
        if (this.goal !== this._spread) this.schedule();
        return;
      }
    }

    this.emit();
    if (this.turning && !(this.turning as Turning).dragging) this.schedule();
  };
}

  return Math.min(SPREAD_COUNT - 1, Math.max(0, Math.round(spread)));

  const s = clampSpread(spread);
  return { left: s === 0 ? null : 2 * s, right: s === LEAF_COUNT ? null : 2 * s + 1 };

<![CDATA[/** Page-turn maths: pure functions, so the feel of the leaf can be tested without a browser. */
export type Direction = 1 | -1;

/** Share of a turn at which letting go finishes it instead of settling back. */
export const COMPLETE_AT = 0.4;
/** Pointer speed (px per ms, toward finishing the turn) that counts as a flick. */
export const FLICK_SPEED = 0.6;
/** How far the pointer travels for a full turn, in page widths. */
const TRAVEL = 1.1;
/** Time constant (ms) of the settle animation. */
const SETTLE_MS = 100;

const clamp01 = (n: number) => Math.min(1, Math.max(0, n));

/**
 * Turn progress (0 = untouched, 1 = finished) from where a drag started and where it is now.
 * A forward turn is dragged leftwards, a backward turn rightwards. `base` lets a re-grab
 * continue from the leaf's current progress instead of jumping.
 */
export function dragProgress(startX: number, x: number, pageWidth: number, direction: Direction, base = 0): number {
  const travelled = direction === 1 ? startX - x : x - startX;
  return clamp01(base + travelled / (pageWidth * TRAVEL));
}

/** Where a released leaf should go. `velocity` is px/ms, positive when it moves toward finishing. */
export function settleTarget(progress: number, velocity: number): 0 | 1 {
  if (velocity >= FLICK_SPEED) return 1;
  if (velocity <= -FLICK_SPEED) return 0;
  return progress >= COMPLETE_AT ? 1 : 0;
}

/** One animation frame of the settle: an ease-out that never overshoots and snaps when it arrives. */
export function stepProgress(progress: number, target: 0 | 1, dtMs: number): number {
  const next = progress + (target - progress) * (1 - Math.exp(-dtMs / SETTLE_MS));
  return Math.abs(target - next) < 0.002 ? target : next;
}

/** The leaf's own turn value: 0 lying on the right, 1 lying on the left. */
export function leafTurn(direction: Direction, progress: number): number {
  return direction === 1 ? progress : 1 - progress;
}

/** A continuous position through the book, e.g. 3.25 is a quarter of the way from spread 3 to 4. */
export function bookPosition(spread: number, direction: Direction, progress: number): number {
  return spread + direction * progress;
}

 * Horizontal offset of the leaves, in percent of the book width. A closed cover sits half a page
 * off-centre so the single page is centred on screen; the offset eases away as the book opens.
 */
export function bookShift(position: number, leafCount: number): number {
  const opening = 1 - clamp01(position);
  const closing = clamp01(position - (leafCount - 1));
  return -25 * opening + 25 * closing + 0;
}

export function tapeCount(position: number, spreadCount: number): number {
  return Math.round(clamp01(position / (spreadCount - 1)) * 999);
